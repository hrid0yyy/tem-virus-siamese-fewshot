\newcommand{\patch}[5]{%

% encoder trapezoid narrowing to the right; \encoder{name}{x}{y}{fill}
\newcommand{\encoder}[4]{%
  \draw[fill=#4, draw=black!70, line width=0.6pt]
     (#2,#3-0.50) -- (#2+1.90,#3-0.26) -- (#2+1.90,#3+0.26) -- (#2,#3+0.50) -- cycle;
  \node[sub] at (#2+0.84,#3) {DenseNet201};
  \coordinate (#1-in) at (#2,#3); \coordinate (#1-out) at (#2+1.90,#3);}

% projection head; \head{name}{x}{y}
\newcommand{\head}[3]{%
  \draw[fill=headfill, draw=black!70, line width=0.6pt]
     (#2,#3-0.30) -- (#2+0.90,#3-0.17) -- (#2+0.90,#3+0.17) -- (#2,#3+0.30) -- cycle;
  \node[sub] at (#2+0.43,#3) {MLP};
  \coordinate (#1-in) at (#2,#3); \coordinate (#1-out) at (#2+0.90,#3);}

% feature vector; \fvec{name}{x}{y}{cells}{height}{fill}
\newcommand{\fvec}[6]{%
  \foreach \i in {1,...,#4} {\draw[fill=#6, draw=black!60, line width=0.3pt]
     (#2-0.08,{#3-#5/2+(\i-1)*#5/#4}) rectangle (#2+0.08,{#3-#5/2+\i*#5/#4});}
  \coordinate (#1-in) at (#2-0.08,#3); \coordinate (#1-out) at (#2+0.08,#3);}

\newcommand{\proto}[2]{\node[star, star points=5, star point ratio=2.3, fill=#2, draw=black!70,
                                line width=0.3pt, inner sep=1.0pt] at #1 {};}
