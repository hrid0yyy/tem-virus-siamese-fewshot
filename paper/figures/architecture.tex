\newcommand{\patch}[5]{%
  \node[inner sep=0pt, draw=#5, line width=1.4pt] (#1) at (#3)

% encoder trapezoid narrowing to the right; \encoder{name}{x}{y}{fill}{text}
\newcommand{\encoder}[5]{%
  \draw[fill=#4, draw=black!70, line width=0.6pt]
     (#2,#3-0.62) -- (#2+1.9,#3-0.32) -- (#2+1.9,#3+0.32) -- (#2,#3+0.62) -- cycle;
  \node[small] at (#2+0.95,#3) {#5};
  \coordinate (#1-in) at (#2,#3); \coordinate (#1-out) at (#2+1.9,#3);}

% projection head; \head{name}{x}{y}
\newcommand{\head}[3]{%
  \draw[fill=headfill, draw=black!70, line width=0.6pt]
     (#2,#3-0.32) -- (#2+1.15,#3-0.18) -- (#2+1.15,#3+0.18) -- (#2,#3+0.32) -- cycle;
  \node[small] at (#2+0.575,#3) {MLP};
  \coordinate (#1-in) at (#2,#3); \coordinate (#1-out) at (#2+1.15,#3);}

% feature vector; \fvec{name}{x}{y}{cells}{height}{fill}
\newcommand{\fvec}[6]{%
  \foreach \i in {1,...,#4} {\draw[fill=#6, draw=black!60, line width=0.3pt]
     (#2-0.09,{#3-#5/2+(\i-1)*#5/#4}) rectangle (#2+0.09,{#3-#5/2+\i*#5/#4});}
  \coordinate (#1-in) at (#2-0.09,#3); \coordinate (#1-out) at (#2+0.09,#3);}

\newcommand{\proto}[2]{\node[star, star points=5, star point ratio=2.3, fill=#2, draw=black!70,
